/**
 * Configuración de conexión a la base de datos MySQL.
 * Se utiliza un Pool de conexiones para optimizar el rendimiento y la reutilización de hilos.
 */

const mysql = require('mysql2/promise'); // Importamos directamente la versión promise para que pool.getConnection() devuelva una promesa y no un callback
require('dotenv').config();

// Objeto de configuración centralizado
const dbConfig = {
  host: process.env.DB_HOST,
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  database: process.env.DB_NAME,
  waitForConnections: true,
  connectionLimit: parseInt(process.env.DB_CONNECTION_LIMIT) || 10, // Escalabilidad configurable
  queueLimit: 0,
  enableKeepAlive: true,
  keepAliveInitialDelay: 10000
};

// Crear el pool
const pool = mysql.createPool(dbConfig);

/**
 * Verifica la disponibilidad de la base de datos al iniciar el servicio.
 * Esta función es invocada por el servidor principal antes de levantar la API.
 */
const verifyConnection = async () => {
  try {
    const connection = await pool.getConnection();
    console.log(`[DB] Conexión establecida exitosamente. ID de hilo: ${connection.threadId}`);
    connection.release();
    return true;
  } catch (error) {
    console.error('[DB] Error crítico: No se pudo conectar a la base de datos.');
    console.error(`[DB] Detalle: ${error.message}`);
    // Si la DB falla, el proceso debería terminar
    process.exit(1); 
  }
};

// Exportamos el pool y la función de verificación
module.exports = {
  pool,
  verifyConnection
};
